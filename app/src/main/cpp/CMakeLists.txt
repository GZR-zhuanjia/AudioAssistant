cmake_minimum_required(VERSION 3.22.1)
project(audioassistant LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(FFMPEG_ROOT "${CMAKE_CURRENT_LIST_DIR}/../ffmpeg/${ANDROID_ABI}")
foreach(component avformat avcodec avfilter avutil swresample)
    if(NOT EXISTS "${FFMPEG_ROOT}/lib/lib${component}.so")
        message(FATAL_ERROR "Missing FFmpeg ${component}. Run scripts/build-ffmpeg.sh before building the app.")
    endif()
    add_library(${component} SHARED IMPORTED)
    set_target_properties(${component} PROPERTIES IMPORTED_LOCATION "${FFMPEG_ROOT}/lib/lib${component}.so")
endforeach()

add_library(audioassistant SHARED engine.cpp jni_bridge.cpp)
target_include_directories(audioassistant PRIVATE "${FFMPEG_ROOT}/include")
target_compile_options(audioassistant PRIVATE -Wall -Wextra -Wpedantic -Werror=return-type)
target_link_options(audioassistant PRIVATE "-Wl,-z,max-page-size=16384")
target_link_libraries(audioassistant PRIVATE avformat avcodec avfilter avutil swresample log)
